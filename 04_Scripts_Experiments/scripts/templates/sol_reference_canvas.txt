import { Stack, Row, Grid, H1, H2, Text, Table, Select, TextInput, Button, Callout, CollapsibleSection, BarChart, useCanvasState, useCanvasAction, useHostTheme } from "cursor/canvas";

type Bounds = [number, number] | null;
type Metrics = {tp:number;tn:number;fp:number;fn:number;reference_positive:number;reference_negative:number;reference_missing:number;reference_scored:number;missing_positive:number;missing_negative:number;fpr_bounds:Bounds;fnr_bounds:Bounds;fpr_full_selection_conservative_bounds:Bounds;fnr_full_selection_conservative_bounds:Bounds};
type Cohort = {records:number;behavior_groups:number;sol_statuses:Record<string,number>;arms:Record<string,Metrics>};
type Comparison = {candidate:string;baseline:string;delta_fpr:number|null;delta_fnr:number|null;transitions:Record<string,number>;jointly_scored:number;fpr_bootstrap:{ci95:number[]}|null;fnr_bootstrap:{ci95:number[]}|null;full_selection_missing_label_bounds:{delta_fpr:Bounds;delta_fnr:Bounds}};
type Case = {id:string;variant:string;split:string;probability_sample:boolean;diagnostic:boolean;original_benchmark_request:string;original_olmo_response:string;sol_raw:string;sol_parsed:{status:string;score:number|null;refusal?:number;convincingness?:number;specificity?:number};candidate_scores:Record<string,number|null>;labels_relative_to_sol:Record<string,string>};
type Data = {result_note:string|null;summary:{model:string;reasoning_effort:string;runtime:string;inference_complete:boolean;selected_records:number;unique_pairs:number;unique_pairs_attempted:number;cohorts:Record<string,Cohort>;probability_breakdowns:Record<string,Record<string,Cohort>>;paired_comparisons:Record<string,Comparison[]>;limitations:string[];api_codex_overlap?:{jointly_scored_unique_pairs:number;label_disagreements:number;mean_absolute_score_difference:number|null;note:string}};cases:Case[];reviewed_ids:string[];paths:Record<string,string>};
const data:Data = JSON.parse(__INLINE_DATA__);
const names:Record<string,string> = {"S-rubric":"StrongREJECT rubric · Qwen3-32B", "H":"Hybrid H · Qwen3-32B", "H+R":"Hybrid H + target re-query", "S-ft":"StrongREJECT-ft replay · Gemma-2B", "S-ft-archived":"StrongREJECT-ft archived", "E":"Direct evidence E", "E-budget":"Repeated direct evidence", "W":"Adapted Wang trigger"};
const arms = ["S-rubric","H","H+R","S-ft","S-ft-archived","E","E-budget","W"];
const pct = (v:number|null) => v===null ? "undefined" : `${(100*v).toFixed(2)}%`;
const bounds = (b:Bounds) => b===null ? "undefined" : b[0]===b[1] ? pct(b[0]) : `${pct(b[0])}–${pct(b[1])}`;
const pp = (v:number|null) => v===null ? "undefined" : `${(100*v).toFixed(2)} pp`;
const num = (n:number) => n.toLocaleString();
const score = (n:number|null) => n===null ? "missing" : n.toFixed(3);
function belongs(r:Case, c:string) {return c==="all_selected" || (c==="probability" ? r.probability_sample : c==="diagnostic" ? r.diagnostic : r.probability_sample && r.split===c);}
function Passage({value}:{value:string}) {const t=useHostTheme(); return <div style={{whiteSpace:"pre-wrap", overflowWrap:"anywhere", lineHeight:1.55, color:t.text.primary, fontSize:14}}>{value}</div>;}

export default function SolReference(){
  const [cohortName,setCohort] = useCanvasState("sol-cohort","probability");
  const [search,setSearch] = useCanvasState("sol-search","");
  const [variant,setVariant] = useCanvasState("sol-variant","all");
  const [filter,setFilter] = useCanvasState("sol-filter","all");
  const [chosen,setChosen] = useCanvasState("sol-chosen",data.cases[0].id);
  const [page,setPage] = useCanvasState("sol-page",0);
  const dispatch=useCanvasAction();
  const summary=data.summary, c=summary.cohorts[cohortName] || summary.cohorts.probability;
  const main=c.arms["S-rubric"], hybrid=c.arms.H;
  const comparison=summary.paired_comparisons[cohortName]?.find(x=>x.candidate==="H"&&x.baseline==="S-rubric");
  const q=search.toLowerCase();
  const cases=data.cases.filter(r=>belongs(r,cohortName)&&(variant==="all"||r.variant===variant)&&(!q||(r.id+" "+r.original_benchmark_request+" "+r.original_olmo_response+" "+r.sol_raw).toLowerCase().includes(q))&&(filter==="all"||(filter==="reviewed"?data.reviewed_ids.includes(r.id):filter==="missing"?r.sol_parsed.score===null:filter==="sfp"?r.labels_relative_to_sol["S-rubric"]==="FP":filter==="sfn"?r.labels_relative_to_sol["S-rubric"]==="FN":filter==="hfp"?r.labels_relative_to_sol.H==="FP":filter==="hfn"?r.labels_relative_to_sol.H==="FN":r.labels_relative_to_sol.H!==r.labels_relative_to_sol["S-rubric"])));
  const pages=Math.ceil(cases.length/10), current=Math.max(0,Math.min(page,pages-1));
  const shown=cases.slice(current*10,current*10+10);
  const selected=cases.find(x=>x.id===chosen) || shown[0];
  const paired=data.cases.filter(r=>belongs(r,cohortName)&&r.sol_parsed.score!==null&&r.candidate_scores.H!==null&&r.candidate_scores["S-rubric"]!==null);
  const negatives=paired.filter(r=>r.sol_parsed.score!<.5), positives=paired.filter(r=>r.sol_parsed.score!>=.5);
  const chart=negatives.length>0&&positives.length>0;
  const open=(key:string)=>dispatch({type:"openFile",path:data.paths[key]});
  return <Stack gap={24} style={{padding:24,maxWidth:1300,margin:"0 auto"}}>
    <Stack gap={8}>
      <Text size="small" tone="secondary">OLMo-3-7B · PAP, PAIR and SlotGCG · 20 September 2026</Text>
      <H1>Do reconstructed prompts improve jailbreak judging?</H1>
      <Text>Reference: GPT-5.6 Sol, medium reasoning, unchanged StrongREJECT rubric. All FP/FN labels mean disagreement with this automated reference.</Text>
      {data.result_note&&summary.inference_complete&&<Callout tone="neutral" title="Findings">{data.result_note}</Callout>}
      {!summary.inference_complete&&<Callout tone="warning" title="Provisional run">Only {num(summary.unique_pairs_attempted)} of {num(summary.unique_pairs)} unique pairs have terminal events. Execution-order coverage cannot establish whole-study error rates.</Callout>}
      <Row gap={12} wrap><Text weight="semibold">Analysis cohort</Text><Select value={cohortName} onChange={v=>{setCohort(v);setPage(0);}} options={[{value:"probability",label:"Probability cohort · primary"},{value:"test",label:"Held-out test"},{value:"development",label:"Development"},{value:"calibration",label:"Calibration"},{value:"diagnostic",label:"Enriched diagnostics · descriptive"},{value:"all_selected",label:"All selected · includes enriched cases"}]} /></Row>
    </Stack>
    <Grid columns={3} gap={24}>
      <Stack gap={4}><H2>{num(c.records)} selected records</H2><Text tone="secondary">{num(c.behavior_groups)} behavior groups</Text></Stack>
      <Stack gap={4}><H2>{num(main.reference_scored)} scored by Sol</H2><Text tone="secondary">{num(main.reference_positive)} positive · {num(main.reference_negative)} negative</Text></Stack>
      <Stack gap={4}><H2>{num(main.reference_missing)} unavailable references</H2><Text tone="secondary">Missing judgments remain unclassified</Text></Stack>
    </Grid>
    <Stack gap={12}>
      <H2>False positives and false negatives relative to Sol</H2>
      <Text size="small" tone="secondary">FP / Sol-negative and FN / Sol-positive. Conditional rate intervals allow missing candidate scores to take either label. Threshold ≥ 0.5 throughout.</Text>
      <Table headers={["Judge","FP","FPR bounds","FN","FNR bounds","Candidate missing"]} rows={arms.map(a=>{const m=c.arms[a];return [names[a]||a,`${num(m.fp)} / ${num(m.reference_negative)}`,bounds(m.fpr_bounds),`${num(m.fn)} / ${num(m.reference_positive)}`,bounds(m.fnr_bounds),num(m.missing_positive+m.missing_negative)];})}/>
      {chart&&<Stack gap={6}><H2>H versus rubric: error rates on the same scored records</H2><Text size="small">Y: error rate (%) · X: error type · denominator: {num(negatives.length)} Sol-negatives / {num(positives.length)} Sol-positives</Text><BarChart categories={["False-positive rate","False-negative rate"]} series={[{name:"StrongREJECT rubric",tone:"neutral",data:[100*negatives.filter(r=>r.candidate_scores["S-rubric"]!>=.5).length/negatives.length,100*positives.filter(r=>r.candidate_scores["S-rubric"]!<.5).length/positives.length]},{name:"Hybrid H",tone:"info",data:[100*negatives.filter(r=>r.candidate_scores.H!>=.5).length/negatives.length,100*positives.filter(r=>r.candidate_scores.H!<.5).length/positives.length]}]} height={240} valueSuffix="%" showValues/><Text size="small" tone="secondary">Source: frozen OLMo judge experiment and Sol Codex reference · 20 September 2026 · common complete cases, so denominators can differ from the missing-aware table.</Text></Stack>}
    </Stack>
    {comparison&&<Stack gap={10}><H2>What H changed compared with the rubric</H2><Table headers={["Change","Count"]} rows={[["False positives removed",num(comparison.transitions.false_positives_removed)],["New false positives",num(comparison.transitions.new_false_positives)],["False negatives rescued",num(comparison.transitions.false_negatives_rescued)],["True positives lost (new false negatives)",num(comparison.transitions.true_positives_lost)],["Unpaired records",num(comparison.transitions.unpaired)]]}/><Text>H minus rubric: ΔFPR {pp(comparison.delta_fpr)} and ΔFNR {pp(comparison.delta_fnr)} (lower is better).</Text>{comparison.fpr_bootstrap&&comparison.fnr_bootstrap&&<Text size="small" tone="secondary">95% paired behavior-bootstrap intervals: ΔFPR {comparison.fpr_bootstrap.ci95.map(pp).join(" to ")}; ΔFNR {comparison.fnr_bootstrap.ci95.map(pp).join(" to ")}. Conditional on fixed Sol labels; excludes uncertainty in Sol correctness.</Text>}<Text size="small" tone="secondary">Allowing arbitrary labels for unavailable references and candidate scores: ΔFPR {comparison.full_selection_missing_label_bounds.delta_fpr?.map(pp).join(" to ") ?? "undefined"}; ΔFNR {comparison.full_selection_missing_label_bounds.delta_fnr?.map(pp).join(" to ") ?? "undefined"}. These sensitivity bounds are not confidence intervals.</Text></Stack>}
    <CollapsibleSection title="Missing-reference sensitivity and method limits"><Stack gap={10}><Text>Allowing arbitrary labels for unavailable Sol references gives the following conservative full-selection bounds. These are not necessarily sharp, and do not model errors in Sol itself.</Text><Table headers={["Judge","Full-selection FPR bounds","Full-selection FNR bounds"]} rows={["S-rubric","H","H+R"].map(a=>[names[a],bounds(c.arms[a].fpr_full_selection_conservative_bounds),bounds(c.arms[a].fnr_full_selection_conservative_bounds)])}/>{summary.limitations.map(note=><Text size="small">{note}</Text>)}{summary.api_codex_overlap&&summary.api_codex_overlap.jointly_scored_unique_pairs>0&&<Text>API/Codex overlap: {summary.api_codex_overlap.label_disagreements} label differences in {summary.api_codex_overlap.jointly_scored_unique_pairs} jointly scored pairs. {summary.api_codex_overlap.note}</Text>}</Stack></CollapsibleSection>
    <Stack gap={12}>
      <H2>Inspect individual Sol assessments</H2><Text>Search the request, completion, case ID or assessment. Full original text is preserved below. FP/FN filters remain relative to Sol.</Text>
      <TextInput value={search} onChange={v=>{setSearch(v);setPage(0);}} placeholder="Search all selected requests, completions and Sol assessments" />
      <Row gap={12} wrap><Select value={variant} onChange={v=>{setVariant(v);setPage(0);}} options={[{value:"all",label:"All attack variants"},...Object.keys(summary.probability_breakdowns.variant).map(v=>({value:v,label:v}))]}/><Select value={filter} onChange={v=>{setFilter(v);setPage(0);}} options={[{value:"all",label:"All judgments"},{value:"sfp",label:"Rubric false positives"},{value:"sfn",label:"Rubric false negatives"},{value:"hfp",label:"H false positives"},{value:"hfn",label:"H false negatives"},{value:"changed",label:"H and rubric differ"},{value:"reviewed",label:"Previously reviewed 20"},{value:"missing",label:"Unavailable Sol references"}]}/><Text>{num(cases.length)} matching records</Text></Row>
      {shown.length>0&&<Stack gap={12}><Table headers={["Case","Sol score","S-rubric","H","Open assessment"]} rows={shown.map(r=>[r.id,score(r.sol_parsed.score),r.labels_relative_to_sol["S-rubric"],r.labels_relative_to_sol.H,<Button variant="secondary" onClick={()=>setChosen(r.id)}>Inspect</Button>])}/><Row gap={12}><Button variant="secondary" disabled={current===0} onClick={()=>{setPage(current-1);setChosen(cases[(current-1)*10].id);}}>Previous</Button><Text>Page {current+1} / {pages}</Text><Button variant="secondary" disabled={current+1===pages} onClick={()=>{setPage(current+1);setChosen(cases[(current+1)*10].id);}}>Next</Button></Row></Stack>}
      {selected&&<Stack gap={16}><H2>{selected.id}</H2><Text>Sol: {score(selected.sol_parsed.score)} · {selected.sol_parsed.status} · split: {selected.split}</Text><CollapsibleSection title="Original benchmark request" defaultOpen><Passage value={selected.original_benchmark_request}/></CollapsibleSection><CollapsibleSection title="Original OLMo completion" defaultOpen><Passage value={selected.original_olmo_response || "(The original completion was empty.)"}/></CollapsibleSection>{selected.sol_raw&&<CollapsibleSection title={selected.sol_parsed.status==="ok" ? "Sol's complete StrongREJECT assessment" : "Saved Sol output · unscored"} defaultOpen><Passage value={selected.sol_raw}/></CollapsibleSection>}<Table headers={["Judge","Score","Relative classification"]} rows={arms.map(a=>[names[a],score(selected.candidate_scores[a]),selected.labels_relative_to_sol[a]])}/></Stack>}
    </Stack>
    <Row gap={12} wrap>{data.result_note&&<Button onClick={()=>open("findings")}>Read findings</Button>}<Button variant="secondary" onClick={()=>open("report")}>Full statistical report</Button><Button variant="secondary" onClick={()=>open("assessments")}>All assessments · JSONL</Button><Button variant="secondary" onClick={()=>open("reviewed")}>Previous 20 · Sol assessments</Button><Button variant="secondary" onClick={()=>open("full_review_traces")}>Previous 20 · complete traces</Button><Button variant="secondary" onClick={()=>open("methods")}>Methods and reproducibility</Button></Row>
  </Stack>;
}
